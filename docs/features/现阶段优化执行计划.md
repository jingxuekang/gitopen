#### Day 4-5: 数据缓存优化

**任务清单：**

1. **实现商品数据缓存**
   
   创建 `miniprogram/utils/cache.js`：
   ```javascript
   // 缓存管理工具
   const CACHE_KEYS = {
     HOME_DATA: 'cache_home_data',
     PRODUCTS: 'cache_products',
     CATEGORIES: 'cache_categories'
   }
   
   const CACHE_EXPIRE = {
     HOME_DATA: 5 * 60 * 1000,      // 5分钟
     PRODUCTS: 10 * 60 * 1000,      // 10分钟
     CATEGORIES: 30 * 60 * 1000     // 30分钟
   }
   
   // 设置缓存
   function setCache(key, data, expire) {
     const cacheData = {
       data,
       timestamp: Date.now(),
       expire: expire || CACHE_EXPIRE[key] || 5 * 60 * 1000
     }
     wx.setStorageSync(key, cacheData)
   }
   
   // 获取缓存
   function getCache(key) {
     try {
       const cacheData = wx.getStorageSync(key)
       if (!cacheData) return null
       
       const now = Date.now()
       if (now - cacheData.timestamp > cacheData.expire) {
         wx.removeStorageSync(key)
         return null
       }
       
       return cacheData.data
     } catch (e) {
       return null
     }
   }
   
   // 清除缓存
   function clearCache(key) {
     if (key) {
       wx.removeStorageSync(key)
     } else {
       // 清除所有缓存
       Object.values(CACHE_KEYS).forEach(k => {
         wx.removeStorageSync(k)
       })
     }
   }
   
   module.exports = {
     CACHE_KEYS,
     setCache,
     getCache,
     clearCache
   }
   ```

2. **在页面中使用缓存**
   
   示例（首页）：
   ```javascript
   const { setCache, getCache, CACHE_KEYS } = require('../../utils/cache')
   
   loadHomeData() {
     // 先尝试从缓存读取
     const cachedData = getCache(CACHE_KEYS.HOME_DATA)
     if (cachedData) {
       this.setData(cachedData)
     }
     
     // 然后请求最新数据
     wx.cloud.callFunction({
       name: 'getHomeData'
     }).then(res => {
       if (res.result.code === 0) {
         const data = res.result.data
         this.setData(data)
         // 更新缓存
         setCache(CACHE_KEYS.HOME_DATA, data)
       }
     })
   }
   ```

3. **实现请求去重**
   
   在 `utils/api.js` 中添加：
   ```javascript
   // 请求队列，防止重复请求
   const requestQueue = new Map()
   
   function callFunction(name, data = {}, options = {}) {
     const requestKey = `${name}_${JSON.stringify(data)}`
     
     // 如果相同请求正在进行，返回该请求的 Promise
     if (requestQueue.has(requestKey)) {
       return requestQueue.get(requestKey)
     }
     
     const promise = wx.cloud.callFunction({ name, data })
       .finally(() => {
         // 请求完成后从队列中移除
         requestQueue.delete(requestKey)
       })
     
     requestQueue.set(requestKey, promise)
     return promise
   }
   ```

### 步骤 3: 缓存实施示例

**首页使用缓存：**

```javascript
// pages/index/index.js
const { setCache, getCache, CACHE_KEYS } = require('../../utils/cache')

Page({
  data: {
    banners: [],
    hotProducts: [],
    newProducts: [],
    groupActivities: [],
    loading: true
  },

  onLoad() {
    this.loadData()
  },

  onPullDownRefresh() {
    this.loadData(true)
  },

  async loadData(forceRefresh = false) {
    // 如果不是强制刷新，先尝试从缓存读取
    if (!forceRefresh) {
      const cachedData = getCache(CACHE_KEYS.HOME_DATA)
      if (cachedData) {
        this.setData({ 
          ...cachedData, 
          loading: false 
        })
      }
    }

    try {
      const res = await wx.cloud.callFunction({
        name: 'getHomeData'
      })

      if (res.result.code === 0) {
        const data = res.result.data
        this.setData({ 
          ...data, 
          loading: false 
        })
        
        // 更新缓存
        setCache(CACHE_KEYS.HOME_DATA, data)
      }
    } catch (error) {
      console.error('加载首页数据失败:', error)
      this.setData({ loading: false })
    } finally {
      wx.stopPullDownRefresh()
    }
  }
})
```
